cd /home/site/wwwroot/middleware-platform && NODE_ENV=production HOME=/home node << 'EOF'
const Database = require('better-sqlite3');
const axios = require('axios');

const dbPath = '/home/middleware-prod.db';
const db = new Database(dbPath);
const API_BASE = 'https://api.doclittle.site';
const merchantId = '6d1daad9-13c7-490c-916d-0ccc02506db5';

console.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
console.log('🧪 Testing Voice Agent Product Search & Checkout');
console.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n');

(async () => {
    try {
        // Test 1: Search all products
        console.log('📦 Test 1: Search All Products');
        const response1 = await axios.post(`${API_BASE}/voice/products/search`, {
            merchant_id: merchantId,
            query: null
        }, { headers: { 'Content-Type': 'application/json' } });

        console.log(`✅ Product Count: ${response1.data.product_count}`);
        if (response1.data.products && response1.data.products.length > 0) {
            console.log(`\n📋 Sample Products (first 5):`);
            response1.data.products.slice(0, 5).forEach((p, idx) => {
                console.log(`  ${idx + 1}. ${p.name || 'NO NAME'} - $${p.price} (Available: ${p.available ? 'Yes' : 'No'}, Inventory: ${p.inventory})`);
            });
        }

        // Test 2: Search with query
        console.log('\n\n🔍 Test 2: Search with Query "blue"');
        const response2 = await axios.post(`${API_BASE}/voice/products/search`, {
            merchant_id: merchantId,
            query: 'blue'
        }, { headers: { 'Content-Type': 'application/json' } });

        console.log(`✅ Product Count: ${response2.data.product_count}`);
        if (response2.data.products && response2.data.products.length > 0) {
            console.log(`\n📋 Matching Products:`);
            response2.data.products.forEach((p, idx) => {
                console.log(`  ${idx + 1}. ${p.name || 'NO NAME'} - $${p.price}`);
            });
        }

        // Test 3: Create checkout
        console.log('\n\n🛒 Test 3: Create Checkout');
        const testProduct = response1.data.products[0];
        if (!testProduct) {
            console.log('❌ No products available for checkout test');
            return;
        }

        console.log(`Product: ${testProduct.name} (${testProduct.id})`);
        
        // Get inventory before
        const productBefore = db.prepare('SELECT inventory FROM products WHERE id = ?').get(testProduct.id);
        const inventoryBefore = productBefore ? productBefore.inventory : 0;
        console.log(`Inventory before: ${inventoryBefore}`);

        const checkoutResponse = await axios.post(`${API_BASE}/voice/checkout/create`, {
            merchant_id: merchantId,
            product_id: testProduct.id,
            quantity: 1,
            customer_name: 'Test Customer',
            customer_phone: '+1234567890',
            customer_email: 'test@example.com'
        }, { headers: { 'Content-Type': 'application/json' } });

        console.log(`✅ Status: ${checkoutResponse.status}`);
        console.log(`✅ Success: ${checkoutResponse.data.success}`);
        console.log(`✅ Checkout ID: ${checkoutResponse.data.checkout_id}`);

        // Get inventory after (should be same, only decrements on payment completion)
        const productAfter = db.prepare('SELECT inventory FROM products WHERE id = ?').get(testProduct.id);
        const inventoryAfter = productAfter ? productAfter.inventory : 0;
        console.log(`\n📊 Inventory after: ${inventoryAfter}`);
        console.log(`ℹ️  Note: Inventory only decrements when payment is completed`);

        // Test 4: Complete checkout (simulate payment)
        if (checkoutResponse.data.checkout_id) {
            console.log('\n\n💳 Test 4: Complete Checkout (Simulate Payment)');
            const completeResponse = await axios.post(
                `${API_BASE}/voice/checkout/complete/${checkoutResponse.data.checkout_id}`,
                {
                    payment_status: 'succeeded',
                    payment_intent_id: 'test_payment_intent_' + Date.now()
                },
                { headers: { 'Content-Type': 'application/json' } }
            );

            console.log(`✅ Status: ${completeResponse.status}`);
            console.log(`✅ Success: ${completeResponse.data.success || completeResponse.data.order_id ? 'Yes' : 'No'}`);

            // Check inventory after payment
            const productAfterPayment = db.prepare('SELECT inventory FROM products WHERE id = ?').get(testProduct.id);
            const inventoryAfterPayment = productAfterPayment ? productAfterPayment.inventory : 0;
            console.log(`\n📊 Inventory after payment: ${inventoryAfterPayment}`);
            
            if (inventoryAfterPayment < inventoryBefore) {
                console.log(`✅ Inventory correctly decremented by 1`);
            } else {
                console.log(`⚠️  Inventory not decremented (check order creation logic)`);
            }
        }

        db.close();
        console.log('\n\n━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
        console.log('✅ All Tests Complete!');
        console.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n');

    } catch (error) {
        console.error('❌ Test Error:', error.message);
        if (error.response) {
            console.error('   Status:', error.response.status);
            console.error('   Data:', JSON.stringify(error.response.data, null, 2));
        }
        db.close();
        process.exit(1);
    }
})();
