    SELECT name FROM sqlite_master 
    WHERE type='table' AND name NOT LIKE 'sqlite_%'
    ORDER BY name
  `).all().map(row => row.name);
};

    try {
      const count = db1.prepare(`SELECT COUNT(*) as count FROM ${tableName}`).get().count;
      db1Status = `✅ ${count} rows`.padEnd(15);
    } catch (e) {
      db1Status = '✅ (error)';
    }
  }
  
  if (inDb2) {
    try {
      const count = db2.prepare(`SELECT COUNT(*) as count FROM ${tableName}`).get().count;
      db2Status = `✅ ${count} rows`.padEnd(15);
    } catch (e) {
      db2Status = '✅ (error)';
    }
  }
  
  console.log(tableName.padEnd(30) + db1Status.padEnd(20) + db2Status);
});

    return [];

    const schema2 = getTableSchema(db2, tableName);
