/**

const subdomain = 'akin-dunbar';
const testEmail = 'akin.dunbar@gmail.com';
const API_BASE = 'https://api.doclittle.site';

let allTestsPassed = true;
const results = {
  tenant: {
    merchant: { passed: false, data: null },
    customer: { passed: false, data: null },
    products: { passed: false, count: 0, samples: [] }
  },
  voice: {
    productSearch: { passed: false, products: [], error: null },
    checkoutCreate: { passed: false, checkout: null, error: null },
    checkoutComplete: { passed: false, order: null, error: null },
    inventory: { passed: false, before: 0, after: 0, decremented: false }
  }

// PART 1: TENANT DATA TEST

console.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');

// Test Merchant
console.log('📋 Testing Merchant...');
try {
  let merchant = null;
  
  try {
    merchant = db.getMerchantBySubdomain(subdomain);
  } catch (error) {
    const allMerchants = db.getAllMerchants();
    merchant = allMerchants.find(m => 
      (m.subdomain && m.subdomain === subdomain) ||
      (m.name && m.name.toLowerCase().includes(subdomain))
    );
  }
  
  if (merchant) {
    results.tenant.merchant.passed = true;
    results.tenant.merchant.data = {
      id: merchant.id,
      name: merchant.name,
      subdomain: merchant.subdomain || 'N/A'
    };
    console.log(`   ✅ Merchant found: ${merchant.name} (ID: ${merchant.id})`);
    console.log(`   📋 Subdomain: ${merchant.subdomain || 'N/A'}`);
  } else {
    console.log(`   ❌ Merchant not found`);
    allTestsPassed = false;
  }
} catch (error) {
  console.log(`   ❌ Error: ${error.message}`);
  allTestsPassed = false;
}

// Test Customer
console.log('\n📧 Testing Customer...');
try {
  const customer = db.getCustomerByEmail(testEmail);
  
  if (customer) {
    results.tenant.customer.passed = true;
    results.tenant.customer.data = {
      id: customer.id,
      email: customer.email,
      name: customer.name,
      merchant_id: customer.merchant_id,
      email_verified: customer.email_verified === 1 || customer.email_verified === true,
      password_hash: customer.password_hash ? 'SET' : 'NOT SET'
    };
    console.log(`   ✅ Customer found: ${customer.name || customer.email}`);
    console.log(`   📧 Email: ${customer.email}`);
    console.log(`   ✅ Verified: ${results.tenant.customer.data.email_verified ? 'Yes' : 'No'}`);
    console.log(`   🔑 Password: ${results.tenant.customer.data.password_hash}`);
    
    if (!results.tenant.customer.data.email_verified) {
      console.log(`   ⚠️  Warning: Email not verified`);
    }
    if (results.tenant.customer.data.password_hash === 'NOT SET') {
      console.log(`   ⚠️  Warning: Password not set`);
    }
  } else {
    console.log(`   ❌ Customer not found`);
    allTestsPassed = false;
  }
} catch (error) {
  console.log(`   ❌ Error: ${error.message}`);
  allTestsPassed = false;
}

// Test Products
console.log('\n📦 Testing Products...');
try {
  if (results.tenant.merchant.data) {
    const merchantId = results.tenant.merchant.data.id;
    const products = db.getProductsByMerchant(merchantId);
    
    results.tenant.products.count = products.length;
    results.tenant.products.samples = products.slice(0, 5).map(p => ({
      id: p.id,
      name: p.name || p.Name || 'Unnamed',
      price: p.price || p.Price || 0,
      inventory: p.inventory || p.Inventory || 0
    }));
    
    if (products.length > 0) {
      results.tenant.products.passed = true;
      console.log(`   ✅ Found ${products.length} products`);
      console.log(`   📦 Sample products:`);
      products.slice(0, 5).forEach((p, i) => {
        const name = p.name || p.Name || 'Unnamed';
        const price = p.price || p.Price || 0;
        const inventory = p.inventory || p.Inventory || 0;
        console.log(`      ${i + 1}. ${name} - $${price} (Inventory: ${inventory})`);
      });
    } else {
      console.log(`   ❌ No products found`);
      allTestsPassed = false;
    }
  } else {
    console.log(`   ❌ Cannot test products - merchant not found`);
    allTestsPassed = false;
  }
} catch (error) {
  console.log(`   ❌ Error: ${error.message}`);
  allTestsPassed = false;
}

// PART 2: VOICE AGENT TEST
// ============================================
console.log('\n\n━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
console.log('2️⃣  TESTING VOICE AGENT');
console.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n');

if (!results.tenant.merchant.data) {
  console.log('❌ Cannot test voice agent - merchant not found');
  allTestsPassed = false;
} else {
  const merchantId = results.tenant.merchant.data.id;
  const testProduct = results.tenant.products.samples[0];
  
  // Test 1: Product Search
  console.log('🔍 Testing Product Search...');
  try {
    const searchResponse = await axios.post(`${API_BASE}/voice/products/search`, {
      merchant_id: merchantId,
      query: testProduct ? testProduct.name.split(' ')[0] : ''
    }, {
      headers: { 'Content-Type': 'application/json' }
    });
    
    if (searchResponse.data && searchResponse.data.success) {
      results.voice.productSearch.passed = true;
      results.voice.productSearch.products = searchResponse.data.products || [];
      console.log(`   ✅ Product search successful`);
      console.log(`   📦 Found ${results.voice.productSearch.products.length} products`);
      
      if (results.voice.productSearch.products.length > 0) {
        const firstProduct = results.voice.productSearch.products[0];
        console.log(`   📋 Sample product:`);
        console.log(`      Name: ${firstProduct.name || 'N/A'}`);
        console.log(`      Price: ${firstProduct.price_spoken || firstProduct.price || 'N/A'}`);
        console.log(`      Available: ${firstProduct.available ? 'Yes' : 'No'}`);
        console.log(`      Inventory: ${firstProduct.inventory || 'N/A'}`);
        
        // Check if product has valid name
        if (!firstProduct.name || firstProduct.name === 'undefined' || firstProduct.name === 'NO NAME') {
          console.log(`   ⚠️  Warning: Product name is invalid: "${firstProduct.name}"`);
        }
      }
    } else {
      console.log(`   ❌ Product search failed: ${searchResponse.data?.error || 'Unknown error'}`);
      allTestsPassed = false;
    }
  } catch (error) {
    results.voice.productSearch.error = error.message;
    console.log(`   ❌ Error: ${error.message}`);
    if (error.response) {
      console.log(`   📋 Response status: ${error.response.status}`);
      console.log(`   📋 Response data: ${JSON.stringify(error.response.data)}`);
    }
    allTestsPassed = false;
  }
  
  // Test 2: Checkout Create
  console.log('\n💳 Testing Checkout Creation...');
  try {
    if (results.voice.productSearch.products.length > 0) {
      const product = results.voice.productSearch.products[0];
      
      // Get current inventory before checkout
      const dbProduct = db.getProduct(product.id);
      results.voice.inventory.before = dbProduct ? (dbProduct.inventory || dbProduct.Inventory || 0) : 0;
      
      const checkoutResponse = await axios.post(`${API_BASE}/voice/checkout/create`, {
        merchant_id: merchantId,
        product_id: product.id,
        quantity: 1,
        customer_phone: '+15551234567',
        customer_name: 'Test Customer',
        customer_email: 'test@example.com'
      }, {
        headers: { 'Content-Type': 'application/json' }
      });
      
      if (checkoutResponse.data && checkoutResponse.data.success) {
        results.voice.checkoutCreate.passed = true;
        results.voice.checkoutCreate.checkout = checkoutResponse.data.checkout;
        console.log(`   ✅ Checkout created successfully`);
        console.log(`   📋 Checkout ID: ${checkoutResponse.data.checkout.id}`);
        console.log(`   💰 Amount: $${checkoutResponse.data.checkout.amount || 'N/A'}`);
        console.log(`   📦 Product: ${checkoutResponse.data.checkout.product_name || 'N/A'}`);
      } else {
        console.log(`   ❌ Checkout creation failed: ${checkoutResponse.data?.error || 'Unknown error'}`);
        allTestsPassed = false;
      }
    } else {
      console.log(`   ⚠️  Skipping checkout test - no products available`);
    }
  } catch (error) {
    results.voice.checkoutCreate.error = error.message;
    console.log(`   ❌ Error: ${error.message}`);
    if (error.response) {
      console.log(`   📋 Response status: ${error.response.status}`);
      console.log(`   📋 Response data: ${JSON.stringify(error.response.data)}`);
    }
    allTestsPassed = false;
  }
  
  // Test 3: Checkout Complete
  console.log('\n✅ Testing Checkout Completion...');
  try {
    if (results.voice.checkoutCreate.checkout) {
      const checkoutId = results.voice.checkoutCreate.checkout.id;
      const productId = results.voice.checkoutCreate.checkout.product_id;
      
      const completeResponse = await axios.post(`${API_BASE}/voice/checkout/complete/${checkoutId}`, {
        payment_method: 'test',
        payment_intent_id: 'test_intent_' + Date.now()
      }, {
        headers: { 'Content-Type': 'application/json' }
      });
      
      if (completeResponse.data && completeResponse.data.success) {
        results.voice.checkoutComplete.passed = true;
        results.voice.checkoutComplete.order = completeResponse.data.order;
        console.log(`   ✅ Checkout completed successfully`);
        console.log(`   📋 Order ID: ${completeResponse.data.order?.id || 'N/A'}`);
        console.log(`   📊 Status: ${completeResponse.data.order?.status || 'N/A'}`);
        
        // Check inventory after checkout
        const dbProductAfter = db.getProduct(productId);
        results.voice.inventory.after = dbProductAfter ? (dbProductAfter.inventory || dbProductAfter.Inventory || 0) : 0;
        results.voice.inventory.decremented = results.voice.inventory.after < results.voice.inventory.before;
        
        console.log(`\n   📦 Inventory Check:`);
        console.log(`      Before: ${results.voice.inventory.before}`);
        console.log(`      After: ${results.voice.inventory.after}`);
        console.log(`      Decremented: ${results.voice.inventory.decremented ? '✅ Yes' : '❌ No'}`);
        
        if (results.voice.inventory.decremented) {
          results.voice.inventory.passed = true;
        } else {
          console.log(`   ⚠️  Warning: Inventory was not decremented`);
        }
      } else {
        console.log(`   ❌ Checkout completion failed: ${completeResponse.data?.error || 'Unknown error'}`);
        allTestsPassed = false;
      }
    } else {
      console.log(`   ⚠️  Skipping checkout completion test - no checkout created`);
    }
  } catch (error) {
    results.voice.checkoutComplete.error = error.message;
    console.log(`   ❌ Error: ${error.message}`);
    if (error.response) {
      console.log(`   📋 Response status: ${error.response.status}`);
      console.log(`   📋 Response data: ${JSON.stringify(error.response.data)}`);

const voicePassed = results.voice.productSearch.passed && results.voice.checkoutCreate.passed && results.voice.checkoutComplete.passed && results.voice.inventory.passed;

if (tenantPassed && voicePassed) {

  console.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
} else {

  console.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  if (!tenantPassed) {

    if (!results.tenant.merchant.passed) console.log('   - Merchant not found');

    if (!results.tenant.products.passed) console.log('   - No products found');
  }
  if (!voicePassed) {
    console.log('\n💡 VOICE AGENT ISSUES:');
    if (!results.voice.productSearch.passed) console.log('   - Product search API failed');
    if (!results.voice.checkoutCreate.passed) console.log('   - Checkout creation failed');
    if (!results.voice.checkoutComplete.passed) console.log('   - Checkout completion failed');
    if (!results.voice.inventory.passed) console.log('   - Inventory not decremented');

}
